Standard library · Databases

db/pg

Imported as import "db/pg" as pg, its names are then pg.…. Every signature below is the one the checker infers.

Types

Conn

A connection: libpq's object.

  • Conn(pg: Ptr)

Table

What a query answered: the column names, the rows, and how many rows the statement touched — inserted, updated, deleted, or for a SELECT the count returned.

  • Table(columns: List(Str), rows: List(Row), affected: Int)

Row

One row, with the column names beside it so a cell can be asked for by name. A cell is None for SQL NULL.

  • Row(columns: List(Str), cells: List(Option(Str)))

Functions

fn connect(conninfo: Str) -> Result(Conn, Str)

A connection from a libpq connection string: key=value pairs, or a postgresql://user@host/db URL, or "" for the defaults and the PGHOST/PGDATABASE/… environment. The error is libpq's, naming what it could not do.

# not run
c = pg.connect("host=localhost dbname=shop user=ada password=secret")?
pg.connect("postgresql://ada@db.example.com/shop")?
pg.connect("host=nowhere.invalid")   # Err: could not translate host name "nowhere.invalid" ...

fn disconnect(c: Conn) -> Unit

Closes the connection and frees it.

# not run
pg.disconnect(c)

fn database(c: Conn) -> Str

The database the connection is to.

# not run
pg.database(c)   # "shop"

fn server_version(c: Conn) -> Int

The server's version as one number: 180004 is 18.4.

# not run
pg.server_version(c) / 10000   # 18

fn library_version() -> Int

libpq's own version, the same way.

pg.library_version() >= 100000   # => true

fn query(c: Conn, sql: Str, params: List(Str)) -> Result(Table, Str)

A query with its parameters as $1, $2, …, each a string the server reads as the column's type. The whole answer, or the server's error.

# not run
t = pg.query(c, "select name, age from people where age > $1", Cons("30", Nil))?
t.columns                                   # Cons(name, Cons(age, Nil))
len(t.rows)                                 # 1
pg.query(c, "select nope", Nil)             # Err(ERROR:  column "nope" does not exist ...)

fn query_opt(c: Conn, sql: Str, params: List(Option(Str))) -> Result(Table, Str)

The same with a None where a parameter is SQL NULL.

# not run
pg.query_opt(c, "insert into people values ($1, $2)", Cons(Some("bob"), Cons(None, Nil)))?

fn exec(c: Conn, sql: Str) -> Result(Table, Str)

SQL as it stands, no parameters — several statements separated by ; are allowed, and the answer is the last one's. For anything with a value from outside in it, query is the one to use.

# not run
pg.exec(c, "create table t (n int); insert into t values (1), (2)")?

fn run(c: Conn, sql: Str, params: List(Str)) -> Result(Int, Str)

A command for its count: rows inserted, updated or deleted.

# not run
pg.run(c, "update people set age = age + 1 where name = $1", Cons("ada", Nil))   # Ok(1)
pg.run(c, "delete from people", Nil)                                            # Ok(2)

fn one(c: Conn, sql: Str, params: List(Str)) -> Result(Option(Row), Str)

The first row, if there is one.

# not run
match pg.one(c, "select * from people where name = $1", Cons("ada", Nil))?
  Some(r) -> pg.int(r, "age")
  None -> -1

fn scalar(c: Conn, sql: Str, params: List(Str)) -> Result(Option(Str), Str)

The first cell of the first row: Ok(None) when there is no row or the cell is NULL.

# not run
pg.scalar(c, "select count(*) from people", Nil)   # Ok(Some(2))
pg.scalar(c, "select now()", Nil)                  # Ok(Some(2026-09-17 01:30:00.123456+03))

fn prepare(c: Conn, name: Str, sql: Str) -> Result(Str, Str)

A statement parsed and planned once under a name, for running many times. The name comes back, so prepare(...)? reads naturally.

# not run
pg.prepare(c, "by_name", "select age from people where name = $1")?
pg.exec_prepared(c, "by_name", Cons("ada", Nil))?

fn exec_prepared(c: Conn, name: Str, params: List(Str)) -> Result(Table, Str)

# not run
t = pg.exec_prepared(c, "by_name", Cons("ada", Nil))?

fn begin(c: Conn) -> Result(Int, Str)

# not run
pg.begin(c)?
pg.run(c, "update accounts set balance = balance - 10 where id = 1", Nil)?
pg.run(c, "update accounts set balance = balance + 10 where id = 2", Nil)?
pg.commit(c)?

fn commit(c: Conn) -> Result(Int, Str)

fn rollback(c: Conn) -> Result(Int, Str)

fn transaction(c: Conn, f: (Conn) -> Result('a, Str)) -> Result('a, Str)

f(c) between a BEGIN and a COMMIT; an Err from f — a failed query, or one of its own — rolls back instead and is the answer.

# not run
move = \c ->
  pg.run(c, "update accounts set balance = balance - 10 where id = 1", Nil)?
  pg.run(c, "update accounts set balance = balance + 10 where id = 2", Nil)
pg.transaction(c, move)   # Ok(1), or the error, and nothing moved

fn in_transaction(c: Conn) -> Bool

Whether a transaction is open on the connection — begin was run and neither commit nor rollback yet, including one the server has marked failed.

# not run
pg.in_transaction(c)   # false

fn escape(c: Conn, s: Str) -> Str

A value quoted for pasting into SQL text, quotes and all — for the places a $1 cannot go, such as COPY options or a SET. Parameters are the right way everywhere else.

# not run
pg.escape(c, "O'Neil")   # 'O''Neil'

fn escape_name(c: Conn, s: Str) -> Str

A name quoted as an identifier: "table name".

# not run
pg.escape_name(c, "my table")   # "my table"

fn first(t: Table) -> Option(Row)

The first row.

pg.first(pg.Table(Cons("n", Nil), Nil, 0))   # => None

fn column_index(t: Table, name: Str) -> Int

A column's position in a table, or -1.

pg.column_index(pg.Table(Cons("name", Cons("age", Nil)), Nil, 0), "age")   # => 1

fn column(t: Table, name: Str) -> List(Option(Str))

One column of the whole table, as the cells.

t = pg.Table(Cons("n", Nil), Cons(pg.Row(Cons("n", Nil), Cons(Some("1"), Nil)), Cons(pg.Row(Cons("n", Nil), Cons(None, Nil)), Nil)), 0)
pg.column(t, "n")   # => Cons(Some(1), Cons(None, Nil))

fn at(r: Row, i: Int) -> Option(Str)

A cell by position: None for NULL and for a position the row does not have.

r = pg.Row(Cons("name", Cons("age", Nil)), Cons(Some("ada"), Cons(None, Nil)))
pg.at(r, 0)   # => Some(ada)
pg.at(r, 1)   # => None
pg.at(r, 5)   # => None

fn field(r: Row, name: Str) -> Option(Str)

A cell by column name; None for NULL and for a name the row does not have.

r = pg.Row(Cons("name", Cons("age", Nil)), Cons(Some("ada"), Cons(Some("36"), Nil)))
pg.field(r, "name")   # => Some(ada)
pg.field(r, "nope")   # => None

fn text(r: Row, name: Str) -> Str

The cell as text, "" for NULL.

r = pg.Row(Cons("name", Cons("age", Nil)), Cons(Some("ada"), Cons(None, Nil)))
pg.text(r, "name")   # => ada
pg.text(r, "age") + "|"   # => |

fn int(r: Row, name: Str) -> Int

The cell as an integer, 0 for NULL or for text that is not one.

r = pg.Row(Cons("age", Nil), Cons(Some("36"), Nil))
pg.int(r, "age")   # => 36

fn float(r: Row, name: Str) -> Float

The cell as a float, 0 for NULL.

r = pg.Row(Cons("price", Nil), Cons(Some("9.5"), Nil))
pg.float(r, "price")   # => 9.5

fn bool(r: Row, name: Str) -> Bool

The cell as a boolean: PostgreSQL sends t and f.

r = pg.Row(Cons("ok", Nil), Cons(Some("t"), Nil))
pg.bool(r, "ok")   # => true

fn null(r: Row, name: Str) -> Bool

Whether the cell is NULL.

r = pg.Row(Cons("age", Nil), Cons(None, Nil))
pg.null(r, "age")   # => true
